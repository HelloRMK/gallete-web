<script>
(function(){
  var BRAIN='{{ section.settings.brain_url | default: "https://remake-ai-web.vercel.app/api/chat" }}';
  var GREET={{ section.settings.greeting | default: "¡Hola! Soy Gally, de Galleté 🍵 ¿Te ayudo con el matcha o con las cookies?" | json }};
  var fab=document.getElementById('lumiFab'),panel=document.getElementById('lumiPanel'),msgs=document.getElementById('lumiMsgs'),
      input=document.getElementById('lumiText'),send=document.getElementById('lumiSend'),x=document.getElementById('lumiX');
  var history=[],opened=false;
  function esc(s){var d=document.createElement('div');d.textContent=s;return d.innerHTML;}
  function linkify(s){return esc(s).replace(/(https?:\/\/[^\s]+)/g,'<a href="$1" target="_blank" rel="noopener">$1</a>');}
  function add(role,txt){var el=document.createElement('div');el.className='lumi-m '+(role==='me'?'me':'bot');el.innerHTML=linkify(txt);msgs.appendChild(el);msgs.scrollTop=msgs.scrollHeight;return el;}
  function open(){panel.classList.add('on');opened=true;if(!history.length){add('bot',GREET);}input.focus();}
  function close(){panel.classList.remove('on');}
  fab.addEventListener('click',function(){panel.classList.contains('on')?close():open();});
  x.addEventListener('click',close);
  async function sendMsg(){
    var t=(input.value||'').trim();if(!t)return;
    input.value='';add('me',t);history.push({role:'user',content:t});
    send.disabled=true;var typing=add('bot','Gally está escribiendo…');typing.classList.add('typing');
    try{
      var r=await fetch(BRAIN,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({bot:'gallete',messages:history.slice(-12)})});
      var d=await r.json();var reply=(d&&d.reply)||'Perdona, tuve un problemita. Escríbenos a galleteusa@gmail.com y te ayudamos 💚';
      typing.remove();add('bot',reply);history.push({role:'assistant',content:reply});
    }catch(e){typing.remove();add('bot','Perdona, no pude responder ahora. Escríbenos a galleteusa@gmail.com 💚');}
    send.disabled=false;input.focus();
  }
  send.addEventListener('click',sendMsg);
  input.addEventListener('keydown',function(e){if(e.key==='Enter'){e.preventDefault();sendMsg();}});
})();
</script>
